cmake_minimum_required(VERSION 3.20)
project(UltraArchStudio VERSION 2.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ULTRAARCH_SANITIZERS "Enable ASan/UBSan in the native core test harness" OFF)
option(ULTRAARCH_REQUIRE_TOOLS "Fail required real-tool tests instead of skipping" ${WIN32})
option(ULTRAARCH_IPO "Enable measured opt-in release IPO/LTO" OFF)
option(ULTRAARCH_FAULT_INJECTION "Enable controlled crash points for developer tests" OFF)
find_package(Threads REQUIRED)
add_library(UltraArchCore STATIC
    src/core/RamCandidate.cpp
    src/core/RamInput.cpp
    src/core/StreamingVerifier.cpp src/core/ResourceValidator.cpp src/core/WorkspaceManager.cpp src/core/PublicationTransaction.cpp src/core/ToolCapabilities.cpp src/core/CompressionSpec.cpp src/core/Hasher.cpp src/core/InputManifest.cpp src/core/ContentAnalyzer.cpp src/core/TaskMatrixGenerator.cpp src/core/ResourceGuard.cpp
    src/core/ProcessManager.cpp src/core/ArchiveAssembler.cpp src/core/EmbeddedTools.cpp
    src/core/OptimizerEngine.cpp src/core/PreprocessorPipeline.cpp src/core/SessionManager.cpp
    src/core/ArchiveExtractor.cpp src/core/PresetManager.cpp)
if(ULTRAARCH_IPO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT ipoAvailable OUTPUT ipoError)
    if(NOT ipoAvailable)
        message(FATAL_ERROR "IPO unavailable: ${ipoError}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    set_property(TARGET UltraArchCore PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
target_include_directories(UltraArchCore PUBLIC src)
if(ULTRAARCH_FAULT_INJECTION)
    target_compile_definitions(UltraArchCore PRIVATE ULTRAARCH_FAULT_INJECTION)
endif()
target_include_directories(UltraArchCore SYSTEM PUBLIC vendor)
target_link_libraries(UltraArchCore PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(UltraArchCore PUBLIC /utf-8 PRIVATE /W4 /MP)
else()
    target_compile_options(UltraArchCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_executable(UltraArchEncodeWorker src/worker/EncodeWorker.cpp resources/test-manifest.rc)
    add_executable(UltraArchExtractWorker src/worker/ExtractWorker.cpp resources/test-manifest.rc)
    set_target_properties(UltraArchExtractWorker PROPERTIES OUTPUT_NAME "extract-worker")
    target_link_libraries(UltraArchExtractWorker PRIVATE UltraArchCore oleaut32 uuid)
    install(TARGETS UltraArchExtractWorker RUNTIME DESTINATION . COMPONENT Runtime)
    set_target_properties(UltraArchEncodeWorker PROPERTIES OUTPUT_NAME "encode-worker")
    target_link_libraries(UltraArchEncodeWorker PRIVATE UltraArchCore oleaut32 uuid)
    install(TARGETS UltraArchEncodeWorker RUNTIME DESTINATION . COMPONENT Runtime)
    target_compile_definitions(UltraArchCore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(UltraArchCore PUBLIC psapi shell32 ole32 shlwapi bcrypt windowscodecs)
    if(MINGW)
        target_link_options(UltraArchCore INTERFACE -static -static-libgcc -static-libstdc++)
    endif()
endif()
if(ULTRAARCH_SANITIZERS)
    if(NOT WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(UltraArchCore PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(UltraArchCore PUBLIC -fsanitize=address,undefined)
    else()
        message(FATAL_ERROR "This sanitizer option requires the native Clang/GCC core harness")
    endif()
endif()
if(WIN32)
    add_executable(UltraArchVerifyWorker src/worker/VerifyWorker.cpp resources/test-manifest.rc)
    set_target_properties(UltraArchVerifyWorker PROPERTIES OUTPUT_NAME "verify-worker")
    if(ULTRAARCH_FAULT_INJECTION)
        target_compile_definitions(UltraArchVerifyWorker PRIVATE ULTRAARCH_FAULT_INJECTION)
    endif()
    target_link_libraries(UltraArchVerifyWorker PRIVATE UltraArchCore oleaut32 uuid)
    install(TARGETS UltraArchVerifyWorker RUNTIME DESTINATION . COMPONENT Runtime)
    add_executable(UltraArchStudio WIN32 src/main.cpp
        src/ui/MainWindow.cpp src/ui/LeaderboardWidget.cpp src/ui/TelemetryWidget.cpp
        vendor/imgui/misc/cpp/imgui_stdlib.cpp vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
        vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp resources/resource.rc)
    target_include_directories(UltraArchStudio PRIVATE vendor/imgui vendor/imgui/backends)
    add_dependencies(UltraArchStudio UltraArchVerifyWorker UltraArchEncodeWorker UltraArchExtractWorker)
    target_link_libraries(UltraArchStudio PRIVATE UltraArchCore d3d11 dxgi d3dcompiler comctl32 comdlg32 uxtheme uuid dwmapi)
    if(MSVC)
        # Both executables embed app.manifest through their resource scripts.
        foreach(target UltraArchStudio UltraArchVerifyWorker UltraArchEncodeWorker UltraArchExtractWorker)
            target_link_options(${target} PRIVATE /MANIFEST:NO)
        endforeach()
    endif()
    install(TARGETS UltraArchStudio RUNTIME DESTINATION . COMPONENT Runtime)
    # Legacy optional codecs are outside the pinned tool manifest and supported matrix.
    install(DIRECTORY bin/ DESTINATION bin COMPONENT Runtime PATTERN "Codecs" EXCLUDE)
    install(FILES README.md THIRD_PARTY.md DESTINATION . COMPONENT Runtime)
    install(DIRECTORY docs/ DESTINATION docs COMPONENT Runtime
        FILES_MATCHING PATTERN "*.md" PATTERN "*.png")
    install(DIRECTORY licenses/ DESTINATION licenses COMPONENT Runtime)
    install(FILES tools/windows-smoke.ps1 tools/windows-worker-smoke.py tools/benchmark.py
        tools/prepare-roadmap-fixtures.py
        tools/ram-benchmark.ps1
        tools/compare-archivers-windows.py
        tools/compression-benchmark-windows.py tools/analyze-archiver-evidence.py tools/extract-process-trace.py
        DESTINATION tools COMPONENT Developer)
    install(FILES vendor/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt COMPONENT Runtime)
    install(FILES vendor/imgui/LICENSE.txt DESTINATION licenses RENAME imgui-MIT.txt COMPONENT Runtime)
else()
    # Development harness only. The supported desktop product is Windows 10/11 x64.
    add_executable(UltraArchStudio src/main.cpp)
    target_include_directories(UltraArchStudio PRIVATE vendor/imgui vendor/imgui/backends)
    target_link_libraries(UltraArchStudio PRIVATE UltraArchCore)
    message(STATUS "Building native CLI/core test harness; the desktop product targets Windows")
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
